class SerpSpider(scrapy.Spider):
    name = "serp"
    allowed_domains = ["amazon.com"]
    start_urls = ["https://www.amazon.com"]

    def __init__(self, *args, **kwargs):
        self.api_key = os.getenv('SERP_API')
        self.url = (
            f"https://serpapi.com/search.json?"
            f"engine=google"
            f"&location=Austin%2C+Texas%2C+United+States"
            f"&google_domain=google.com"
            f"&gl=us"
            f"&hl=en"
            f"&api_key={self.api_key}"
        )
        self.terms = amazon_queries.terms

    def start_requests(self):
        for term in self.terms:
            yield scrapy.http.Request(
                self.url + f"""&q=site:amazon.com '{urllib.parse.quote(term)}'""",
                method="GET",
                callback=self.parse,
                errback=self.errback,
                encoding="utf-8",
                meta={"term": term},
            )
